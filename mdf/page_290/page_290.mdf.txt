\lx нёра̄
\hm 2
\ps И
\gn кунжа (рыба)

\lx нёра̄-мӣ
\sn 1
\gn идти впереди, ведя за собою других
\sn 2
\ps Урм
\gn начинать первым (что-л. делать)
\sn 3
\ps Алд, Учр
\gn запевать
\et [Эвен нёгара-, Ульч ёро-, Нан меѳреа- идти впереди]

\lx нёрави
\ps П-Т, Нрч, Чмк
\gn мужчина (молодой)
\cf нёрӣ
\et [Сол ниро, Нег неяви, Эвен няри мужчина; Орок няри, Ороч, Удэ ни, Нан наи, най человек; Удэ няула подросток; Ульч няудякан парень]

\lx нёрагин
\ps Учр
\gn передовой олень (в упряжке)

\lx нёра̄дерӣ
\sn 1
\gn идущий впереди и ведущий за собою
\sn 2
\gn передовой
\xv нёра̄дерӣ колхоз
\xn передовой колхоз

\lx нёралав-мӣ
\ps Н
\gn прижать уши (о животных)
\xv сӣрин нёралавча̄л
\xn его уши прижаты

\lx нёра̄мды
\sn 1
\ps З, Учр, Урм, Сх
\gn передовой олень (или собака) в упряжке
\sn 2
\ps И, С-Б фольк
\gn вожак
\cf нёра̄мнӣ
\et [Эвен негамнга, нэвулан, Ороч нерамды, Ульч нгерамды]

\lx нёра̄мнгӯ
\ps Тнг, З, Алд, Учр, Чмк
\cf нёра̄мнӣ

\lx нёра̄мнӣ
\ps П-Т, Н, Е
\sn 1
\ps фольк
\gn вождь, передовой
\sn 2
\gn вожак, знающий дорогу и прокладывающий путь другим
\sn 3
\ps Алд, Учр
\gn запевала
\cf нёра̄мды, нёра̄мнгӯ

\lx нербэкэ̄
\ps Н, Е, И, З
\sn 1
\gn холм, поросший сосновым бором
\sn 2
\gn холм с вырубленным бором
\sn 3
\gn Нербэкэ̄, Нербэкэ̄чэ̄н посёлок Ербогачен
\cf мэрбэкэ̄

\lx нёрӣ
\ps вин. п. нёрӣва
\sn 1
\ps П-Т, Н, Е, И, С
\gn женский посох для езды на олене
\sn 2
\ps С-Б
\gn верхний крюк лыжной палки (для цепляния при подъёмах, для раскапывания снега)
\sn 3
\ps Алд
\gn крюк женского посоха (им направляют сползающие вьюки на оленях)
\cf нёрӯ

\lx нёрӣ
\ps А
\gn мужчина
\cf нёрави

\lx нериё
