\lx либгэ
\sn 1
\gn П-Т, З, Алд первый снежок
\cf ливгэ
\sn 2
\gn Урм, Сх большой снегопад весною
\sn 3
\gn П-Т снежная зима

\lx либгэ-мӣ
\sn 1
\gn П-Т, З, Алд падать, идти (о первом снеге)
\xv либгэрэн
\xn снег пошёл
\sn 2
\gn Урм, Сх валить (о большом снеге весною)
\cf ливгэ-мӣ

\lx ливакича-мӣ
\gn Урм
\cf ливан-мӣ

\lx ливан-мӣ
\gn Н высунуть язык (о человеке)
\cf ливакича-мӣ

\lx ливган
\gn П-Т, Н, Е ледяной затор
\cf левган

\lx ливган-мӣ
\hm 1
\sn 1
\gn П-Т, Н, Е, И, Учр подавиться, поперхнуться
\sn 2
\gn застрять в горле
\cf либган-мӣ, лапкан-мӣ II
\et Нег либган-, Эвен нибган-, Орок нунгбэ-, Ульч, Нан лунгэбэ-
\eg подавиться, проглотить

\lx ливган-мӣ
\hm 2
\gn П-Т, Н, Е, Урм, Сх нагромоздиться (о льде)
\xv ливганэн
\xn набило льду (в устье и на повороте реки)
\cf левган-мӣ
\et Эвен нибган-, Ульч ливги-, Нан лунгбэ-

\lx ливганмука
\sn 1
\gn П-Т, Н, Е, И, Учр, Алд глотка, горло
\sn 2
\gn горлышко посуды
\cf либганмука

\lx ливганчэ-мӣ
\gn Н, Е быть, образоваться (о ледяном заторе)
\xv ливганчэрэн
\xn образовался ледяной затор
\cf левганчэ-мӣ

\lx ливгэ
\gn Н, Е, С-Б, Урм, Сх первый снежок, первопутье
\cf либгэ 1

\lx ливгэ-мӣ
\gn Н, Е, С-Б, Урм, Сх падать, идти (о первом снеге)
\xv ливгэдерэн
\xn идет первый снег
\cf либгэ-мӣ

\lx лигдырэн-мӣ

\lx лигилин

\lx лигирин

\lx липки-мӣ
\ps (3 л. липкӣрэн)
\sn 1
\gn заткнуть, закупорить
\sn 2
\gn защемить, прищемить
\sn 3
\gn преградить путь
\sn 4
\gn потерять голос (от страха)
\sn 5
\gn притупиться (о слухе); оглохнуть; стать тугим на ухо
\sn 6
\gn З занести, замести следы (снегом)
\xv удялин липкирэ
\xn его следы замело
\et Нег, Орок липку-, Удэ ликпа-, Ульч лакпа-, Ороч липпи-
\eg заткнуть

\lx липкив-мӣ
\ps (3 л. липкӣврэн)
\sn 1
\gn закупориться
\sn 2
\gn быть защемлённым
\sn 3
\gn пропасть (о голосе от страха)
\xv дылганми нгэлэдук липкивран
\xn у меня (от страха) пропал голос
\sn 4
\gn притупиться (о слухе)
\xv сӯрви липкивра
\xn у меня притупился слух, заложило уши
\sn 5
\gn быть заложенным (о носе)
\xv онгоктов липкӣврэн
\xn у меня заложило нос
