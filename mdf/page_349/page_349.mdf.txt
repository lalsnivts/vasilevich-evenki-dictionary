\lx сивакӣ
\sn 1
\gn П-Т, Н, Алд, Учр, Урм, Сх деревянный клин
\sn 2
\gn стамеска
\sn 3
\gn зубило
\sn 4
\gn Сивакӣ название реки
\cf шивакӣ, шиволан, севакӣ

\lx сивакта
\sn 1
\gn П-Т, Учр дорожный знак (мох, заложенный за ветку дерева)
\sn 2
\gn Сивакта название реки

\lx сиварӣвӯн
\gn П-Т, Н зубило, слесарный инструмент
\cf хиварӣвӯн

\lx сиварин
\gn П-Т серединка лука для стрельбы (то место, за которое обычно держат лук)

\lx сизарка
\gn А жгут, которым смазывают полозья нарты при спусках

\lx сивачā
\gn З ястреб
\cf кирэктэ

\lx сизи-мӣ
\ps (3 л. сивиран)
\sn 1
\gn П-Т слюнки текут
\lt зачесаться — о губах перед едой
\sn 2
\gn Учр, Урм, Сх испачкать губы чем-либо жирным
\sn 3
\gn разг. обслюнявиться

\lx сизивъе
\gn Тнг барак

\lx сивигэ
\gn Тнг волк
\cf иргичӣ

\lx сивидэ-мӣ
\gn Учр шептать
\cf сивута-мӣ

\lx сивидэденэ
\gn Учр шёпотом
\cf сивутадяна

\lx сивие
\gn А медведь
\cf хомōтӣ

\lx сивилука
\gn Тнг речка
\cf биракāн

\lx сивина
\gn Учр, Урм, Чмк, Сх анат. ложное ребро

\lx сивинā-мӣ
\gn Урм, Сх чувствовать колотьё в боку (от бега, быстрой ходьбы)

\lx сивинчэ
\gn Алд смущённый
\cf хāлдетчэрӣ

\lx сивичā
\gn Учр, Сх замарашка (о ребёнке с постоянно запачканным ртом)

\lx сивси
\gn Олкм, З, Алд, Хнг, Учр, Урм кузнечик

\lx сивси-мӣ
\sn 1
\gn П-Т, Н, Е, З, Алд, Хнг, Учр, Урм трещать (о кузнечиках)
\sn 2
\gn В-Л шептать
\cf сивутā-мӣ

\lx сивсикӣ
\gn Олкм, З, Алд, Хнг, Учр, Урм, Чмк кузнечик

\lx сивсикӯн
\gn П-Т, Н, Е, С-Б, Брг, Урм, Сх степная кобылка (насекомое)
\cf шившикӯн, шиушикӣ

\lx сивсичилин
\gn В-Л шептун

\lx сивулин
\va сивун Н
\gn З, Алд шорох

\lx сивуна-мӣ
\sn 1
\gn Н нашёптывать, наговаривать
\sn 2
\gn шептаться

\lx сивута-мӣ
\gn Н, Тнг, З, Учр, Урм, Сх шептать
\cf сивидэ-мӣ, сивси-мӣ, сивчи-мӣ, хивута-мӣ

\lx сизутадяна
\gn Н, З, Тнг, Учр, Урм, Сх шёпотом
\cf сивидэденэ

\lx сивчи-мӣ
\gn В-Л шептать
\cf сивута-мӣ

\lx сӣвчэ
\sn 1
\gn П-Т, Н, Тнг, З, Алд, Учр, Урм, Сх потушенный, погашенный
\sn 2
\gn охрипший (о голосе)
\cf сӣпчэ

\lx сивэг
\sn 1
\gn П-Т, Е, З, Учр, Урм, Сх заболоченная лужайка, где растёт хвощ
\sn 2
\gn Сивэг название реки
\cf сивэги, сивэр

\lx сивэг-мӣ
\va сивэвэ-мӣ С-Б
\gn Учр пастись на полянке, где растёт хвощ (об оленях)
\cf сивэгӣ-мӣ

\lx сивэги
\gn Тнг
\cf сивэг

\lx сивэгӣ-мӣ
\gn Тнг
\cf сивэг-мӣ

\lx сивэктэ
\sn 1
\gn П-Т, Е, З, Алд, Учр, Урм, Сх хвощ
\sn 2
\gn молодая травка на заболоченных лужайках
\sn 3
\gn С-Б трава „гусятник“
\sn 4
\gn Сивэктэ название реки

\lx сивэр
\gn Брг заболоченная лужайка в тайге, где растёт хвощ
\cf сивэг

\lx сигарка
\gn Тнг нюхательный табак

\lx сигарка-мӣ
\gn Тнг нюхать табак

\lx сигдывӯн
\ps (вин. п. сигдывӯнмэ)
\gn Н, Урм, Сх деревянный гвоздь
\cf хигдывӯн

\lx сигдылэ
\sn 1
\gn П-Т, Н, Брг, Нрч, Алд, Учр, Урм расстояние, промежуток между двумя участками земли
\sn 2
\gn промежуток времени
\sn 3
\gn Тнг, З, Сх узкая долина
\sn 4
\gn З мост
\sn 5
\gn Сх в продолжение пути
\sn 6
\gn Тнг противоположная сторона
\xv бега сигдылэн
\xn новолуние
\cf сигдэлэ̄, хигдылэ̄, тыгдэлэ̄н
