\gn вéрхний

\lx угӣр-мӣ
\ps П-Т, Н, Е
\gn (3 л. угӣррэн) подня́ть вверх, в вóздух
\cf Ср. увӣр-мӣ, увэр-мӣ, уйр-мӣ
\et [Эвен угур-, Удэ уиндэ- подня́ть]

\sn 1
\gn (3 л. угӣрӣврэн) подня́ться вверх, в вóздух
\sn 2
\gn возвыша́ться

\et [Эвен угир-, Удэ уиндэ- подня́ть]

\sn 2
\ps Урм, Сх
\gn приподня́ть (что-л.)

\lx угӣскӣ
\gn вверх, в вóздух
\cf Ср. уйскӣ, увискӣ
\et [Сол угэшхи, Нег увиски, Эвен угэски, Орок уйсэи, Удэ уихи, Ороч, Ульч, Нан уйси вверх; Ман вэсихун вверх]

\lx угитмэр

\lx угитмэрит
\gn вы́ше (о действии)
\xv тар самолёт угитмэрит дэгдерэн
\xn этот самолет леги́т вы́ше

\lx угла-мӣ

\sn 2
