\ps Брг
\gn почитать (кого-л.)
\cf кэне-мӣ 3

\lx талга-мӣ
\hm 2
\ps Ткм
\gn жарить
\cf далга-мӣ, ирй-мӣ 2

\lx талган
\hm 1
\ps Урм
\gn ручка двери

\lx талган
\hm 2
\sn 1
\ps П-Т, И, С, Ткм, Брг, Нрч
\va талгана
\ps С
\gn мука
\cf бурдук
\sn 2
\ps С-Б
\gn печёный хлеб
\cf колобо
\sn 3
\ps В-Л
\gn хлеб (зерно)
\et Бур талха(н)

\ps Учр, Урм, Сх
\gn чаща с валёжником (где водится соболь)

\ps Урм
\gn дёргать за ручку двери

\lx тāлгу-мӣ

\xv тāлгурэн
\xn а) ветер стих; б) Е безветрие

\lx тāлгурā
\ps И
\gn ясный тихий день; ясная тихая погода

\lx талды-мӣ
\ps Сх
\sn 1

\sn 2
\gn гоняться (за зверьком — о собаках)

\lx талды-мӣ
\ps Нрч
\gn ехать вместе с кем-либо
\cf нгэнэлдӣ-мӣ

\lx талдыкта-мӣ
\ps Сх
\gn бегать (от собаки — о зверьке)

\lx тāлй

\lx талив-мӣ
\ps Урм
\gn покрыться туманом, затуманиться

\lx таливӯн
\ps П-Т, Н, Е, И, З, Учр, Урм
\sn 1

\sn 2
\gn береста, щепки для зажигания при лучении рыбы

\lx талинут-мӣ
\ps П-Т, Н, Е, З, Алд, Учр, Урм, Сх

\sn 1
\gn 1) Брг молния; 2) С-Б гром; см. агды 1

\gn 1) молния сверкнула; 2) С-Б гром прогремел; см. агдыран
