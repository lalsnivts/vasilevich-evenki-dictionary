\lx токто-мӣ
\ps 3 л. токторон
\va П-Т, Н, Е, И, С-Б, З, Алд, Учр, Урм, Чмк, Сх
\sn 1
\gn рубить (топором); вырубить; отрубить; отсечь
\sn 2
\gn резать (ножом)
\sn 3
\gn стучать
\et Нег, Ороч, Удэ токто- рубить; долбить; Ульч туктури- стучать палкой
