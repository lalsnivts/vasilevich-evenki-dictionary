\sn 2

\sn 3
\gn сожжённый, обожжённый; обжёгшийся

\lx дегдэ
\hm 1
\gn пожар
\cf дегды
\et Нег, Эвен дегдэ гарь; Удэ, Ульч дегдэ, Нан дегдэсэ

\lx дегдэ
\hm 2
\ps И
\gn оспа (болезнь)

\lx дегдэ-мӣ
\sn 1
\gn гореть
\sn 2
\gn загорать
\sn 3
\gn обжигать
\et Нег, Эвен, Ороч, Удэ, Ульч дегдэ-, Нан дегдэу- гореть; Бур дэгжэ- разгораться

\lx дегдэвкэн-мӣ
\gn поджечь

\lx дегдэк
\ps Сх
\gn гарь, выгоревшее место в лесу
\cf дегдэннэ

\lx дегдэкӯн

\sn 1

\sn 2
\gn Дегдэкӯн название реки

\lx дегдэливкэн-мӣ

\gn зажечь, разжечь

\lx дегдэмнэ

\va дегдэндэ И, Уир
\cf дегдэннэ

\lx дегдэннэ
\ps П-Т, Н, Е, З, Алд
\sn 1
\gn гарь, выгоревшее место в лесу
\sn 2
\gn Дегдэннэ название реки
\cf дегдэк, дегдэмнэ, дегдэндэ, дегдэрӣ, дегдэрнэ

\lx дегдэрӣ
\ps Урм
\va дегдэрнэ Тнг
\cf дегдэннэ

\lx дегин
\hm 1
\sn 1
\gn левый
\xv дегин бэе
\xn З левая сторона
\sn 2
\gn неловкий (о человеке)
\sn 3
\gn левша
\cf девин I

\lx дегин
\hm 2
\ps Н, Ткм, Алд
\gn девять
\cf егин

\lx дегин-дяр
\ps Н, Ткм, Алд, Сх
\gn девяносто
\cf егин-дяр

\lx дегиннгидэ
\gn левая сторона
\cf дегэннгидэ

\lx дегиннгидэдӯ
\gn с прит. суфф. слева (от чего-л., от кого-л. находиться)
\xv дегиннгидэдӯв
\xn слева от меня
\xv дегиннгидэдӯн
\xn слева от него

\lx дегиннгидэдук
\gn с прит. суфф. с левой стороны (прийти)
\xv дегиннгидэдукин дагамам
\xn я подошёл к нему слева

\lx дегиннгидэлӣ
\gn с прит. суфф. влево (передвигаться, двигаться)
\xv дегиннгидэлӣв нгэнэчэтын
\xn оний проходили с левой стороны от меня

\lx дегиннгидэлэ
\sn 1
\gn влево, налево (идти)
\xv дегиннгидэлэ сурукэл!
\xn иди налево!
\sn 2
\gn слева, налево (находиться)
\xv таткит дегиннгидэлэн,кунгакардевкичитынбисин
\xn налево от школы находится детская столовая

\lx дегиннгит
\gn с левой стороны (подойти)

\lx дегиннгӯ
\gn левый
\xv дегиннгӯ нгалэ
\xn левая рука
\cf деиннгӯ

\lx дегинтыкӣ

\cf дегинты

\lx дёгор
\ps Н, Е, В-Л, Олкм, Тнг, З, Уир
\sn 1
\gn нужда, бедность, нищета страдание от лишений
\sn 2
\gn Олкм бедный
\sn 3
\gn Урм, Сх мало
\cf адыкӯн
\cf дёгорӣ
\et Сол дёгор бескормица, гололедица; Онг дёгорон страдание; Нег, Ороч дёбой недостаток, нужда; Удэ дёбо трудно; Ульч дёчо нехватка; дёбо- иметь неудачи в работе; Нан дёбо бедный; дёбон труд; Монг джоба- мучить

\lx дёгор-мӣ
\ps П-Т, Н, Е, Вл, Тнг, З, Урм, Сх
\va дёгот-мӣ
\gn нуждаться, нищенствовать, страдать (от лишений)
\cf дёво-мӣ, дёго-мӣ

\lx дёгорӣ
\sn 1
\gn бедняк, нищий
\sn 2
\gn В-Л, Втм, Нрч нужда
\sn 3
\gn нуждающийся, бедный
\cf дёгор

\lx дёгорӣвӯн
\gn нужда, бедность
